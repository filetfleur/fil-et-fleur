        function getCreationsFiltrees() {
            const recherche = document.getElementById('recherche').value.toLowerCase();

                return 0;

        function mettreAJourStats() {
            document.getElementById('statTotal').textContent = creations.length;
            document.getElementById('statEnVente').textContent = creations.filter(c => c.statut === 'en-vente').length;
            const vendus = creations.filter(c => c.statut === 'vendu');

        function afficherStatsDetaillees() {
            const statutDiv = document.getElementById('chartStatut');
            const total = creations.length || 1;
            const statuts = [
                { cle: 'en-vente', label: 'En vente', cls: 'amber' },
                { cle: 'en-cours', label: 'En cours', cls: 'blue' },
                { cle: 'vendu', label: 'Vendus', cls: 'green' }
            ];
            statutDiv.innerHTML = statuts.map(s => {
                const n = creations.filter(c => c.statut === s.cle).length;
                const pct = Math.round(n / total * 100);
                return `<div class="chart-bar"><span class="chart-bar-label">${s.label}</span><div class="chart-bar-fill ${s.cls}" style="width:${Math.max(pct, 3)}%">${n}</div></div>`;
            }).join('');

            const moisDiv = document.getElementById('chartMois');
            const ventes = creations.filter(c => c.statut === 'vendu' && c.dateVente);
            const parMois = {};
            ventes.forEach(v => {
                const d = new Date(v.dateVente);
                const cle = `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,'0')}`;
                parMois[cle] = (parMois[cle] || 0) + (parseFloat(v.prix) || 0);
            });
            const moisTries = Object.keys(parMois).sort().slice(-6);
            const maxMois = Math.max(...Object.values(parMois), 1);
            moisDiv.innerHTML = moisTries.length === 0
                ? '<p style="color:var(--text-muted);font-size:0.9em;">Aucune vente enregistrée.</p>'
                : moisTries.map(cle => {
                    const [y, m] = cle.split('-');
                    const label = new Date(y, m-1, 1).toLocaleDateString('fr-FR', { month: 'short', year: '2-digit' });
                    const pct = Math.round(parMois[cle] / maxMois * 100);
                    return `<div class="chart-bar"><span class="chart-bar-label">${label}</span><div class="chart-bar-fill pink" style="width:${Math.max(pct, 8)}%">${parMois[cle].toFixed(0)} €</div></div>`;
                }).join('');

            const topDiv = document.getElementById('topPrix');
            const top = [...creations].filter(c => c.prix).sort((a,b) => parseFloat(b.prix) - parseFloat(a.prix)).slice(0, 5);
            topDiv.innerHTML = top.length === 0 ? '<p style="color:var(--text-muted);font-size:0.9em;">Aucune création avec prix.</p>'
                : top.map(c => `<div class="list-item"><strong>${echapper(c.nom)}</strong><span>${parseFloat(c.prix).toFixed(2)} €</span></div>`).join('');

            const derniereDiv = document.getElementById('dernieresVentes');
            const dernieres = creations.filter(c => c.statut === 'vendu' && c.dateVente).sort((a,b) => new Date(b.dateVente) - new Date(a.dateVente)).slice(0, 5);
            derniereDiv.innerHTML = dernieres.length === 0 ? '<p style="color:var(--text-muted);font-size:0.9em;">Aucune vente enregistrée.</p>'
                : dernieres.map(c => `<div class="list-item"><strong>${echapper(c.nom)}</strong><span>${formaterDate(c.dateVente)} · ${parseFloat(c.prix||0).toFixed(2)} €</span></div>`).join('');
        }

        function exporterDonnees() {
            if (creations.length === 0) { alert('Aucune donnée à exporter.'); return; }
            const dataStr = JSON.stringify(creations, null, 2);
            const blob = new Blob([dataStr], { type: 'application/json' });
            const url = URL.createObjectURL(blob);
            const lien = document.createElement('a');
            lien.href = url;
            lien.download = `fil-et-fleur-${new Date().toISOString().slice(0,10)}.json`;
            document.body.appendChild(lien);
            lien.click();
            document.body.removeChild(lien);
            URL.revokeObjectURL(url);
        }

        function importerDonnees(event) {
            const file = event.target.files[0];
            if (!file) return;
            const reader = new FileReader();
            reader.onload = e => {
                try {
                    const data = JSON.parse(e.target.result);
                    if (!Array.isArray(data)) throw new Error('Format invalide');
                    if (!confirm(`Importer ${data.length} création(s) ? Cela remplacera tes données actuelles.`)) {
                        event.target.value = '';
                        return;
                    }
                    creations = data;
                    sauvegarderStorage();
                    afficherCreations();
                    mettreAJourStats();
                    afficherStatsDetaillees();
                    alert('✅ Import réussi !');
                } catch (err) {
                    alert('❌ Fichier invalide : ' + err.message);
                }
                event.target.value = '';
            };
            reader.readAsText(file);
        }

            let photoBase = photoActuelle || document.getElementById('photoUrl').value.trim();
            if (!photoBase) { statutDiv.textContent = '⚠️ Ajoute d\'abord une photo de la peluche.'; statutDiv.style.color = 'var(--danger)'; return; }
            if (!prompt) { statutDiv.textContent = '⚠️ Décris le fond souhaité.'; statutDiv.style.color = 'var(--danger)'; return; }
            if (!AGNES_API_KEY || AGNES_API_KEY === 'YOUR_API_KEY') { statutDiv.textContent = '⚠️ Clé API Agnes non configurée (édite le fichier HTML).'; statutDiv.style.color = 'var(--danger)'; return; }
            statutDiv.textContent = '⏳ Génération en cours... (10-30 s)';
            statutDiv.style.color = 'var(--info)';
            try {
                const response = await fetch(AGNES_API_URL, {
                    method: 'POST',
                    headers: { 'Authorization': `Bearer ${AGNES_API_KEY}`, 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        model: 'agnes-image-2.0-flash',
                        prompt: `Place this crochet plush on the following background: ${prompt}. Keep the plush exactly as in the input image, unchanged, realistic lighting, high quality, centered composition.`,
                        size: '1024x1024',
                        image: [photoBase],
                        extra_body: { response_format: 'url' }
                    })
                });
                if (!response.ok) {
                    let msg = `Erreur HTTP ${response.status}`;
                    try { const e = await response.json(); msg = e.error?.message || e.message || msg; } catch(_) {}
                    throw new Error(msg);
                }
                const data = await response.json();
                const img = data.data?.[0]?.url || data.data?.[0]?.b64_json;
                if (!img) throw new Error('Pas d\'image renvoyée.');
                const finalUrl = img.startsWith('http') ? img : 'data:image/png;base64,' + img;
                const nomFichier = `peluche-${nom.replace(/[^a-z0-9]/gi, '-')}-fond-${Date.now()}.png`;
                telechargerImage(finalUrl, nomFichier);
                statutDiv.textContent = `✅ Image générée et téléchargée : ${nomFichier}`;
                statutDiv.style.color = 'var(--success)';
                if (remplacerPhoto) {
                    if (finalUrl.startsWith('http')) { document.getElementById('photoUrl').value = finalUrl; photoActuelle = ''; }
                    else { photoActuelle = finalUrl; document.getElementById('photoUrl').value = ''; }
                    document.getElementById('previewContainer').innerHTML = `<img src="${finalUrl}" alt="Aperçu">`;
                    statutDiv.textContent += ' — Photo mise à jour.';
                }
            } catch (err) {
                console.error(err);
                statutDiv.textContent = `❌ ${err.message}`;
                statutDiv.style.color = 'var(--danger)';
            }
        }

        function telechargerImage(url, nomFichier) {
            const lien = document.createElement('a');
            lien.href = url;
            lien.download = nomFichier;
            document.body.appendChild(lien);
            lien.click();
            document.body.removeChild(lien);
        }

        function echapper(texte) {
            const div = document.createElement('div');
            div.textContent = texte || '';
            return div.innerHTML;
        }

        function formaterDate(dateStr) {
            const d = new Date(dateStr);
            return d.toLocaleDateString('fr-FR', { day: '2-digit', month: 'short', year: 'numeric' });
        }
